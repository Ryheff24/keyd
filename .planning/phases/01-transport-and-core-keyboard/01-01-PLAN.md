```swift
/// All messages sent over the keyd transport, in both directions.
/// IMPORTANT: This enum defines the wire protocol for Phase 1.
/// New message types may be added in later phases but existing
/// type codes must never be renumbered — doing so breaks the protocol.
public enum KeydMessage: Codable, Sendable {

    // MARK: iOS → Mac messages

    /// A single character or key event from the iOS keyboard
    case keyEvent(KeyEvent)

    /// A modifier+key shortcut from the iOS hotkey panel (e.g. Cmd-C)
    case hotkeyEvent(HotkeyEvent)

    /// First-time pairing: iOS sends its Curve25519 public key
    case pairingRequest(PairingRequest)

    case pairingResponse(PairingResponse)

    case secureInputChanged(SecureInputChanged)

    /// [Phase 2] Reflects macOS field content and cursor state back to iOS
    case reflectState(ReflectState)

    /// [Phase 3] Notifies iOS of frontmost macOS application change
    case appStateChanged(AppStateChanged)
```

```swift
import Foundation

/// Encodes a KeydMessage into a length-prefixed binary frame.
///
/// Frame format: [4-byte big-endian UInt32 length][JSON payload bytes]
/// The receiver reads 4 bytes to get the length, then exactly that many payload bytes.
public final class MessageEncoder: Sendable {
    private let jsonEncoder = JSONEncoder()

    public init() {}

    /// Encode a message into a transport frame.
    /// - Returns: 4-byte length prefix + JSON payload
    public func encode(_ message: KeydMessage) throws -> Data {
        let payload = try jsonEncoder.encode(message)
        let length = UInt32(payload.count).bigEndian
        var frame = withUnsafeBytes(of: length) { Data($0) }
        frame.append(payload)
        return frame
    }
}
```

```swift
import Foundation

/// Decodes length-prefixed binary frames into KeydMessage values.
///
/// Maintains an internal streaming buffer. Feed received bytes via `receive(_:)`.
/// Complete messages are delivered via the `onMessage` callback.
///
/// Thread safety: Not thread-safe. Call from a single serial queue or actor.
public final class MessageDecoder {
    private let jsonDecoder = JSONDecoder()
    private var buffer = Data()

    /// Called with each fully decoded message.
    public var onMessage: ((KeydMessage) -> Void)?

    /// Called when a frame contains invalid data (decode error).
    public var onError: ((Error) -> Void)?

    public init() {}

    /// Feed received bytes into the decoder.
    /// Emits zero or more complete messages via `onMessage`.
    public func receive(_ data: Data) {
        buffer.append(data)
        processBuffer()
    }

    private func processBuffer() {
        // Process as many complete frames as are available
        while buffer.count >= 4 {
            // Read the 4-byte big-endian length prefix
            let lengthBytes = buffer.prefix(4)
            let length = lengthBytes.withUnsafeBytes { ptr in
                UInt32(bigEndian: ptr.load(as: UInt32.self))
            }
            let totalFrameSize = 4 + Int(length)

            // Wait for the full frame payload
            guard buffer.count >= totalFrameSize else { break }

            // Extract the payload
            let payload = buffer[4..<totalFrameSize]

            // Advance buffer past this frame
            buffer = Data(buffer.dropFirst(totalFrameSize))

            // Decode and emit
            do {
                let message = try jsonDecoder.decode(KeydMessage.self, from: payload)
                onMessage?(message)
            } catch {
                onError?(error)
                // Continue processing remaining frames (don't abort on one bad frame)
            }
        }
    }

    /// Reset internal buffer state (e.g., on reconnect).
    public func reset() {
        buffer = Data()
    }
}
```

```swift
import XCTest
@testable import SharedProtocol

final class MessageRoundTripTests: XCTestCase {
    let encoder = MessageEncoder()
    var decoder: MessageDecoder!
    var received: [KeydMessage] = []

    override func setUp() {
        decoder = MessageDecoder()
        received = []
        decoder.onMessage = { [weak self] msg in self?.received.append(msg) }
    }

    // MARK: - Single message round-trips

    func testKeyEventRoundTrip() throws {
        let original = KeydMessage.keyEvent(KeyEvent(unicodeScalar: 97, keyDown: true, modifiers: []))
        let frame = try encoder.encode(original)
        decoder.receive(frame)
        XCTAssertEqual(received.count, 1)
        if case .keyEvent(let ev) = received[0] {
            XCTAssertEqual(ev.unicodeScalar, 97)
            XCTAssertTrue(ev.keyDown)
        } else {
            XCTFail("Wrong message type")
        }
    }

    func testHotkeyEventRoundTrip() throws {
        let original = KeydMessage.hotkeyEvent(HotkeyEvent(virtualKeyCode: VirtualKey.c, keyDown: true, modifiers: .command))
        let frame = try encoder.encode(original)
        decoder.receive(frame)
        XCTAssertEqual(received.count, 1)
        if case .hotkeyEvent(let ev) = received[0] {
            XCTAssertEqual(ev.virtualKeyCode, VirtualKey.c)
            XCTAssertEqual(ev.modifiers, .command)
        } else {
            XCTFail("Wrong message type")
        }
    }

    func testSecureInputChangedRoundTrip() throws {
        let original = KeydMessage.secureInputChanged(SecureInputChanged(isActive: true))
        let frame = try encoder.encode(original)
        decoder.receive(frame)
        XCTAssertEqual(received.count, 1)
        if case .secureInputChanged(let ev) = received[0] {
            XCTAssertTrue(ev.isActive)
        } else {
            XCTFail("Wrong message type")
        }
    }

    // MARK: - Streaming / fragmentation

    func testFragmentedDelivery() throws {
        let msg = KeydMessage.keyEvent(KeyEvent(unicodeScalar: 65, keyDown: false, modifiers: .shift))
        let frame = try encoder.encode(msg)

        // Split the frame in half
        let mid = frame.count / 2
        decoder.receive(frame.prefix(mid))
        XCTAssertEqual(received.count, 0, "Should not emit until full frame received")
        decoder.receive(frame.dropFirst(mid))
        XCTAssertEqual(received.count, 1, "Should emit after receiving remaining bytes")
    }

    func testMultipleCoalescedMessages() throws {
        let msgs: [KeydMessage] = [
            .keyEvent(KeyEvent(unicodeScalar: 104, keyDown: true)),
            .keyEvent(KeyEvent(unicodeScalar: 105, keyDown: true)),
            .keyEvent(KeyEvent(unicodeScalar: 106, keyDown: true)),
        ]
        var combined = Data()
        for msg in msgs {
            combined.append(try encoder.encode(msg))
        }
        decoder.receive(combined)
        XCTAssertEqual(received.count, 3)
    }

    func testDecoderResetClearsBuffer() throws {
        let frame = try encoder.encode(.keyEvent(KeyEvent(unicodeScalar: 97, keyDown: true)))
        decoder.receive(frame.prefix(2))  // Only partial frame
        decoder.reset()
        decoder.receive(frame)  // Full fresh frame after reset
        XCTAssertEqual(received.count, 1)
    }
}
```
