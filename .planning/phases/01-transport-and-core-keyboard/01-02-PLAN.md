```swift
@main
struct KeydAgentApp: App {
    @NSApplicationDelegateAdaptor(AppDelegate.self) var appDelegate

    var body: some Scene {
        // No windows — accessory app lives entirely in menu bar
        Settings { EmptyView() }
    }

final class AppDelegate: NSObject, NSApplicationDelegate {
    var statusBarController: StatusBarController?
    var transportManager: TransportManager?

    func applicationDidFinishLaunching(_ notification: Notification) {
        // Hide from Dock — accessory app only
        NSApp.setActivationPolicy(.accessory)

        // Register as Login Item (idempotent — safe to call every launch)
        try? SMAppService.mainApp.register()

        // Check permissions before accepting any connections
        let checker = PermissionChecker()
        let perms = checker.checkPermissions()
        if !perms.accessibility || !perms.inputMonitoring {
            checker.showPermissionAlert(accessibility: perms.accessibility, inputMonitoring: perms.inputMonitoring)
            // Do not start transport until permissions granted
            // The alert has an "Open System Settings" button; app stays running
            // Re-check on next launch (user must relaunch after granting)
        }

        // Initialize status bar
        statusBarController = StatusBarController()

        // Start transport if permissions are OK
        if perms.accessibility && perms.inputMonitoring {
            startTransport()
        }
    }

    private func startTransport() {
        let injector = InjectionRouter()
        let manager = TransportManager(injectionRouter: injector, statusBar: statusBarController!)
        self.transportManager = manager
        Task { await manager.start() }
    }
}
```

```swift
import Foundation
import Accessibility

struct PermissionResult {
    let accessibility: Bool
    let inputMonitoring: Bool
}

final class PermissionChecker {
    func checkPermissions() -> PermissionResult {
        // Check Accessibility — pass kAXTrustedCheckOptionPrompt:true to auto-show system prompt
        let axOptions: CFDictionary = [kAXTrustedCheckOptionPrompt: true] as CFDictionary
        let axTrusted = AXIsProcessTrustedWithOptions(axOptions)

        // Check Input Monitoring (required since macOS 10.15 for CGEventPost)
        let imTrusted = IOHIDRequestAccess(kIOHIDRequestTypeListenEvent)

        return PermissionResult(accessibility: axTrusted, inputMonitoring: imTrusted)
    }

    func showPermissionAlert(accessibility: Bool, inputMonitoring: Bool) {
        var missing: [String] = []
        if !accessibility { missing.append("Accessibility") }
        if !inputMonitoring { missing.append("Input Monitoring") }

        let alert = NSAlert()
        alert.messageText = "KeydAgent requires permissions"
        alert.informativeText = "Missing permissions: \(missing.joined(separator: ", ")).\n\nGrant both in System Settings → Privacy & Security.\n\nNote: Both permissions are revoked on every Xcode rebuild — toggle off and on again after each rebuild."
        alert.addButton(withTitle: "Open System Settings")
        alert.addButton(withTitle: "Quit")
        alert.alertStyle = .warning

        let response = alert.runModal()
        if response == .alertFirstButtonReturn {
            // Deep-link to Privacy & Security
            if let url = URL(string: "x-apple.systempreferences:com.apple.preference.security?Privacy") {
                NSWorkspace.shared.open(url)
            }
        } else {
            NSApp.terminate(nil)
        }
    }
}
```

```swift
import Network
import Foundation

/// Listens on localhost:12345 for iOS connections forwarded through usbmuxd.
/// When the iOS device is connected via USB, usbmuxd forwards the device's TCP
/// connection to this local port transparently.
actor USBListener {
    static let port: UInt16 = 12345
    private var listener: NWListener?
    var onConnection: ((NWConnection) -> Void)?

    func start() throws {
        let params = NWParameters.tcp
        // Only accept connections from loopback (usbmuxd tunnel only)
        params.requiredLocalEndpoint = NWEndpoint.hostPort(
            host: "127.0.0.1",
            port: NWEndpoint.Port(rawValue: Self.port)!
        )

        let l = try NWListener(using: params, on: NWEndpoint.Port(rawValue: Self.port)!)
        l.newConnectionHandler = { [weak self] connection in
            Task {
                await self?.onConnection?(connection)
            }
        }
        l.stateUpdateHandler = { state in
            switch state {
            case .ready:
                print("[USBListener] Listening on port \(Self.port)")
            case .failed(let error):
                print("[USBListener] Failed: \(error)")
            default:
                break
            }
        }
        l.start(queue: .global(qos: .userInteractive))
        self.listener = l
    }

    func stop() {
        listener?.cancel()
        listener = nil
    }
}
```

```swift
/// Advertises a Bonjour service (_keyd._tcp) and accepts peer connections over AWDL.
/// AWDL is an Apple proprietary peer-to-peer protocol active when Bonjour services
/// are discovered between nearby Apple devices.
actor AWDLListener {
    private var listener: NWListener?
    var onConnection: ((NWConnection) -> Void)?

        // applicationService parameters enable AWDL-capable peer connections
        let params = NWParameters.applicationService
        let l = try NWListener(using: params)
        // Advertise _keyd._tcp Bonjour service — iOS NWBrowser discovers this
        l.service = NWListener.Service(type: "_keyd._tcp")

                await self?.onConnection?(connection)
```

```swift
import Network
import Foundation
import SharedProtocol

/// Connection state for the macOS agent.
enum AgentConnectionState {
    case disconnected
    case usbConnected(NWConnection)
    case awdlConnected(NWConnection)
}

/// Manages USB and AWDL listeners; routes decoded messages to InjectionRouter.
/// Maintains connection state and updates the status bar.
@MainActor
final class TransportManager {
    private let injectionRouter: InjectionRouter
    private let statusBar: StatusBarController
    private var connectionState: AgentConnectionState = .disconnected

    private var usbListener = USBListener()
    private var awdlListener = AWDLListener()

    // One decoder per active connection
    private var activeDecoder: MessageDecoder?
    private var activeEncoder: MessageEncoder?

    // Secure input polling
    private var secureInputTimer: Timer?
    private var lastSecureInputState = false

    init(injectionRouter: InjectionRouter, statusBar: StatusBarController) {
        self.injectionRouter = injectionRouter
        self.statusBar = statusBar
    }

    func start() async {
        do {
            // Start both listeners concurrently (AWDL is pre-warmed per TRANS-04)
            usbListener.onConnection = { [weak self] conn in
                Task { @MainActor in
                    await self?.handleNewConnection(conn, transport: .usb)
                }
            }
            awdlListener.onConnection = { [weak self] conn in
                Task { @MainActor in
                    await self?.handleNewConnection(conn, transport: .awdl)
                }
            }
            try await usbListener.start()
            try await awdlListener.start()
            print("[TransportManager] Both listeners started")
        } catch {
            print("[TransportManager] Failed to start listeners: \(error)")
        }
    }

    private func handleNewConnection(_ connection: NWConnection, transport: ConnectionKind) async {
        print("[TransportManager] New \(transport) connection")

        // If USB connects while AWDL is active, promote USB
        // If AWDL connects while USB is active, keep as backup (pre-warm only)
        switch (connectionState, transport) {
        case (.awdlConnected(let old), .usb):
            // Promote USB, demote AWDL
            old.cancel()
            connectionState = .usbConnected(connection)
            statusBar.update(state: .usb)
        case (.disconnected, .usb), (.disconnected, .awdl):
            // First connection
            connectionState = transport == .usb
                ? .usbConnected(connection)
                : .awdlConnected(connection)
            statusBar.update(state: transport == .usb ? .usb : .awdl)
        case (.usbConnected, .awdl):
            // AWDL comes up while USB is active — keep as fallback but don't promote yet
            // The iOS side also maintains both; USB takes priority
            print("[TransportManager] AWDL pre-warmed while USB active")
            return
        default:
            break
        }

        // Set up decoder for this connection
        let decoder = MessageDecoder()
        let encoder = MessageEncoder()
        self.activeDecoder = decoder
        self.activeEncoder = encoder

        decoder.onMessage = { [weak self] message in
            Task { @MainActor in
                self?.injectionRouter.route(message)
            }
        }

        // Start receiving data
        receiveLoop(on: connection, decoder: decoder, transport: transport)

        // Start secure input polling
        startSecureInputPolling()
    }

    private func receiveLoop(on connection: NWConnection, decoder: MessageDecoder, transport: ConnectionKind) {
        connection.receive(minimumIncompleteLength: 1, maximumLength: 65536) { [weak self] data, _, isComplete, error in
            if let data = data, !data.isEmpty {
                decoder.receive(data)
            }
            if isComplete || error != nil {
                print("[TransportManager] Connection closed: \(error?.localizedDescription ?? "EOF")")
                Task { @MainActor in
                    self?.handleDisconnect(transport: transport)
                }
                return
            }
            // Continue receiving
            self?.receiveLoop(on: connection, decoder: decoder, transport: transport)
        }
    }

    private func handleDisconnect(transport: ConnectionKind) {
        print("[TransportManager] \(transport) disconnected")
        stopSecureInputPolling()

        switch (connectionState, transport) {
        case (.usbConnected, .usb):
            connectionState = .disconnected
            statusBar.update(state: .disconnected)
            // TODO: Phase 1 — if AWDL was pre-warmed, promote it here (TRANS-03)
        case (.awdlConnected, .awdl):
            connectionState = .disconnected
            statusBar.update(state: .disconnected)
        default:
            break
        }
    }

    func send(_ message: KeydMessage) {
        guard let encoder = activeEncoder else { return }
        guard let frame = try? encoder.encode(message) else { return }
        guard let connection = activeConnection else { return }
        connection.send(content: frame, completion: .idempotent)
    }

    private var activeConnection: NWConnection? {
        switch connectionState {
        case .usbConnected(let c): return c
        case .awdlConnected(let c): return c
        case .disconnected: return nil
        }
    }

    // MARK: - Secure Input Polling (AGENT-06)

    private func startSecureInputPolling() {
        stopSecureInputPolling()
        secureInputTimer = Timer.scheduledTimer(withTimeInterval: 0.5, repeats: true) { [weak self] _ in
            Task { @MainActor in
                self?.pollSecureInput()
            }
        }
    }

    private func stopSecureInputPolling() {
        secureInputTimer?.invalidate()
        secureInputTimer = nil
    }

    private func pollSecureInput() {
        let isActive = IsSecureEventInputEnabled()
        if isActive != lastSecureInputState {
            lastSecureInputState = isActive
            let msg = KeydMessage.secureInputChanged(SecureInputChanged(isActive: isActive))
            send(msg)
        }
    }
}

enum ConnectionKind {
    case usb, awdl
}
```

```swift
import CoreGraphics
import Foundation
import SharedProtocol

/// Injects keystrokes into the frontmost macOS application via CGEvent.
/// This is the ONLY write path for text and shortcuts — AX text writes
/// silently fail in Safari, Firefox, and Messages.
final class CGEventInjector {
    private let source = CGEventSource(stateID: .hidSystemState)

    /// Inject a Unicode character as key-down + key-up events.
    func injectCharacter(scalar: UInt32, keyDown: Bool, modifiers: ModifierFlags) {
        guard let unicodeScalar = Unicode.Scalar(scalar),
              let cgFlags = cgFlags(from: modifiers) else { return }

        let event = CGEvent(keyboardEventSource: source, virtualKey: 0, keyDown: keyDown)
        guard let event = event else { return }

        // Use keyboardSetUnicodeString to inject by Unicode — bypasses keyboard layout mapping
        let utf16 = [unicodeScalar].flatMap { $0.utf16 }
        event.keyboardSetUnicodeString(stringLength: utf16.count, unicodeString: utf16)
        event.flags = cgFlags
        event.post(tap: .annotatedSession)
    }

    /// Inject a virtual key code event (Return, Tab, Escape, Backspace, arrow keys, shortcuts).
    /// ALWAYS use defer for modifier key-up to prevent stuck modifier keys (SHORT-11).
    func injectVirtualKey(virtualKeyCode: UInt16, keyDown: Bool, modifiers: ModifierFlags) {
        guard let cgFlags = cgFlags(from: modifiers) else { return }
        let hasModifier = !modifiers.isEmpty

        if keyDown && hasModifier {
            // Send modifier key-down
            sendModifierDown(modifiers)
            defer { sendModifierUp(modifiers) }  // CRITICAL: always release, even on error paths

            // Send the key with modifier
            let keyEvent = CGEvent(keyboardEventSource: source, virtualKey: CGKeyCode(virtualKeyCode), keyDown: true)
            keyEvent?.flags = cgFlags
            keyEvent?.post(tap: .annotatedSession)

            let keyUpEvent = CGEvent(keyboardEventSource: source, virtualKey: CGKeyCode(virtualKeyCode), keyDown: false)
            keyUpEvent?.flags = cgFlags
            keyUpEvent?.post(tap: .annotatedSession)
            // defer fires modifier key-up here
        } else {
            // Simple key with no modifier (or key-up only)
            let event = CGEvent(keyboardEventSource: source, virtualKey: CGKeyCode(virtualKeyCode), keyDown: keyDown)
            event?.flags = cgFlags
            event?.post(tap: .annotatedSession)
        }
    }

    // MARK: - Private helpers

    private func sendModifierDown(_ modifiers: ModifierFlags) {
        if modifiers.contains(.command) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x37, keyDown: true) // kVK_Command
            e?.post(tap: .annotatedSession)
        }
        if modifiers.contains(.shift) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x38, keyDown: true) // kVK_Shift
            e?.post(tap: .annotatedSession)
        }
        if modifiers.contains(.option) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x3A, keyDown: true) // kVK_Option
            e?.post(tap: .annotatedSession)
        }
        if modifiers.contains(.control) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x3B, keyDown: true) // kVK_Control
            e?.post(tap: .annotatedSession)
        }
    }

    private func sendModifierUp(_ modifiers: ModifierFlags) {
        if modifiers.contains(.command) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x37, keyDown: false)
            e?.post(tap: .annotatedSession)
        }
        if modifiers.contains(.shift) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x38, keyDown: false)
            e?.post(tap: .annotatedSession)
        }
        if modifiers.contains(.option) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x3A, keyDown: false)
            e?.post(tap: .annotatedSession)
        }
        if modifiers.contains(.control) {
            let e = CGEvent(keyboardEventSource: source, virtualKey: 0x3B, keyDown: false)
            e?.post(tap: .annotatedSession)
        }
    }

    /// Send a "release all modifiers" sequence — called on transport disconnect (SHORT-11).
    func releaseAllModifiers() {
        for keyCode: CGKeyCode in [0x37, 0x38, 0x3A, 0x3B] {  // Cmd, Shift, Option, Control
            let e = CGEvent(keyboardEventSource: source, virtualKey: keyCode, keyDown: false)
            e?.flags = []
            e?.post(tap: .annotatedSession)
        }
    }

    private func cgFlags(from modifiers: ModifierFlags) -> CGEventFlags? {
        var flags: CGEventFlags = []
        if modifiers.contains(.command)  { flags.insert(.maskCommand) }
        if modifiers.contains(.shift)    { flags.insert(.maskShift) }
        if modifiers.contains(.option)   { flags.insert(.maskAlternate) }
        if modifiers.contains(.control)  { flags.insert(.maskControl) }
        if modifiers.contains(.capsLock) { flags.insert(.maskAlphaShift) }
        return flags
    }
}
```
