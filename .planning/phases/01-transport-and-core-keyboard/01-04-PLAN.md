```swift
import Foundation
import CryptoKit
import Security

/// Manages the macOS side of the one-time QR pairing handshake.
/// Generates a Curve25519 key pair; encodes public key in QR payload;
/// completes ECDH exchange with iOS; stores derived shared secret in Keychain.
final class PairingManager {
    struct PairingPayload: Codable {
        let publicKeyBase64: String   // macOS Curve25519 public key
        let port: UInt16              // usbmuxd TCP port (12345)
        let agentID: String           // macOS machine identifier
        let version: Int              // Protocol version (1)
    }

    private var privateKey: Curve25519.KeyAgreement.PrivateKey?
    private(set) var isPaired = false
    private(set) var pairedDeviceID: String?

    private let keychainService = "com.keyd.agent"
    private let keychainSharedSecretKey = "shared-secret"

    init() {
        // Check if already paired (restore shared secret from Keychain)
        if let _ = loadSharedSecretFromKeychain() {
            isPaired = true
        }
    }

    /// Generate a new keypair and return the QR code payload string.
    func generatePairingQRPayload() -> String {
        let key = Curve25519.KeyAgreement.PrivateKey()
        self.privateKey = key
        let publicKeyData = key.publicKey.rawRepresentation

        let payload = PairingPayload(
            publicKeyBase64: publicKeyData.base64EncodedString(),
            port: 12345,
            agentID: UUID().uuidString,
            version: 1
        )
        let jsonData = try! JSONEncoder().encode(payload)
        return String(data: jsonData, encoding: .utf8)!
    }

    /// Complete pairing when iOS sends its public key.
    /// Returns true if pairing succeeded.
    func completePairing(iOSPublicKeyData: Data, deviceID: String) -> Bool {
        guard let privateKey = privateKey else { return false }

        do {
            let iOSPublicKey = try Curve25519.KeyAgreement.PublicKey(rawRepresentation: iOSPublicKeyData)
            let sharedSecret = try privateKey.sharedSecretFromKeyAgreement(with: iOSPublicKey)
            let symmetricKey = sharedSecret.hkdfDerivedSymmetricKey(
                using: SHA256.self,
                salt: "keyd-v1".data(using: .utf8)!,
                sharedInfo: Data(),
                outputByteCount: 32
            )

            // Store shared secret in Keychain
            saveSharedSecretToKeychain(symmetricKey.withUnsafeBytes { Data($0) })
            isPaired = true
            pairedDeviceID = deviceID
            print("[PairingManager] Paired with device \(deviceID)")
            return true
        } catch {
            print("[PairingManager] Pairing failed: \(error)")
            return false
        }
    }

    // MARK: - Keychain

    private func saveSharedSecretToKeychain(_ secret: Data) {
        let query: [CFString: Any] = [
            kSecClass: kSecClassGenericPassword,
            kSecAttrService: keychainService,
            kSecAttrAccount: keychainSharedSecretKey,
            kSecValueData: secret
        ]
        SecItemDelete(query as CFDictionary)
        SecItemAdd(query as CFDictionary, nil)
    }

    private func loadSharedSecretFromKeychain() -> Data? {
        let query: [CFString: Any] = [
            kSecClass: kSecClassGenericPassword,
            kSecAttrService: keychainService,
            kSecAttrAccount: keychainSharedSecretKey,
            kSecReturnData: true,
            kSecMatchLimit: kSecMatchLimitOne
        ]
        var result: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &result)
        return status == errSecSuccess ? (result as? Data) : nil
    }
}
```

```swift
            img.addRepresentation(rep)
            qrImage = img
        }
```

```swift
import Foundation
import CryptoKit
import Security

/// Manages the iOS side of the one-time QR pairing handshake.
/// Decodes QR payload; generates iOS Curve25519 key; completes ECDH;
/// stores shared secret in Keychain for future auto-connections.
@MainActor
final class PairingManager: ObservableObject {
    @Published var isPaired = false
    @Published var isPairing = false
    @Published var pairingError: String? = nil

    private let keychainService = "com.keyd.app"
    private let keychainSharedSecretKey = "shared-secret"
    private let keychainMacAgentKey = "mac-agent-public-key"

    struct QRPayload: Codable {
        let publicKeyBase64: String
        let port: UInt16
        let agentID: String
        let version: Int
    }

    init() {
        isPaired = loadSharedSecretFromKeychain() != nil
    }

    /// Decode QR code string and complete pairing with macOS agent.
    /// The transport connection must already be established before calling this.
    func completeQRPairing(qrString: String, transportManager: TransportManager) async {
        isPairing = true
        pairingError = nil

        do {
            // Decode QR payload
            guard let qrData = qrString.data(using: .utf8),
                  let payload = try? JSONDecoder().decode(QRPayload.self, from: qrData) else {
                throw PairingError.invalidQRCode
            }

            guard payload.version == 1 else {
                throw PairingError.unsupportedVersion(payload.version)
            }

            guard let macPublicKeyData = Data(base64Encoded: payload.publicKeyBase64) else {
                throw PairingError.invalidQRCode
            }

            // Generate iOS key pair
            let iOSPrivateKey = Curve25519.KeyAgreement.PrivateKey()
            let iOSPublicKey = iOSPrivateKey.publicKey

            // Send pairing request to macOS
            let request = PairingRequest(
                publicKeyData: iOSPublicKey.rawRepresentation,
                deviceID: UIDevice.current.identifierForVendor?.uuidString ?? UUID().uuidString
            )
            transportManager.send(.pairingRequest(request))

            // Complete ECDH
            let macPublicKey = try Curve25519.KeyAgreement.PublicKey(rawRepresentation: macPublicKeyData)
            let sharedSecret = try iOSPrivateKey.sharedSecretFromKeyAgreement(with: macPublicKey)
            let symmetricKey = sharedSecret.hkdfDerivedSymmetricKey(
                using: SHA256.self,
                salt: "keyd-v1".data(using: .utf8)!,
                sharedInfo: Data(),
                outputByteCount: 32
            )

            // Store in Keychain
            let secretData = symmetricKey.withUnsafeBytes { Data($0) }
            saveSharedSecretToKeychain(secretData)
            saveSharedSecretToKeychain(macPublicKeyData, key: keychainMacAgentKey)

            isPaired = true
            isPairing = false
            print("[PairingManager] Pairing complete")
        } catch {
            isPairing = false
            pairingError = error.localizedDescription
            print("[PairingManager] Pairing failed: \(error)")
        }
    }

    // MARK: - Keychain

    private func saveSharedSecretToKeychain(_ data: Data, key: String? = nil) {
        let accountKey = key ?? keychainSharedSecretKey
        let query: [CFString: Any] = [
            kSecClass: kSecClassGenericPassword,
            kSecAttrService: keychainService,
            kSecAttrAccount: accountKey,
            kSecValueData: data
        ]
        SecItemDelete(query as CFDictionary)
        SecItemAdd(query as CFDictionary, nil)
    }

    private func loadSharedSecretFromKeychain() -> Data? {
        let query: [CFString: Any] = [
            kSecClass: kSecClassGenericPassword,
            kSecAttrService: keychainService,
            kSecAttrAccount: keychainSharedSecretKey,
            kSecReturnData: true,
            kSecMatchLimit: kSecMatchLimitOne
        ]
        var result: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &result)
        return status == errSecSuccess ? (result as? Data) : nil
    }

    enum PairingError: LocalizedError {
        case invalidQRCode
        case unsupportedVersion(Int)
        case connectionNotReady

        var errorDescription: String? {
            switch self {
            case .invalidQRCode: return "Invalid QR code. Please scan the code shown by the KeydAgent app on your Mac."
            case .unsupportedVersion(let v): return "Unsupported protocol version \(v)."
            case .connectionNotReady: return "Not connected to Mac. Ensure USB cable is connected and KeydAgent is running."
            }
        }
    }
}
```

```swift
/// UIViewRepresentable wrapper for AVFoundation QR code scanner.
struct QRScannerRepresentable: UIViewRepresentable {
    let onQRCode: (String) -> Void

    func makeUIView(context: Context) -> QRScannerView {
        let view = QRScannerView()
        view.onQRCode = onQRCode
        return view
    }

    func updateUIView(_ uiView: QRScannerView, context: Context) {}
}

/// UIView that uses AVCaptureSession to scan QR codes.
final class QRScannerView: UIView, AVCaptureMetadataOutputObjectsDelegate {
    var onQRCode: ((String) -> Void)?
    private let session = AVCaptureSession()
    private var previewLayer: AVCaptureVideoPreviewLayer?
    private var hasScanned = false

    override init(frame: CGRect) {
        super.init(frame: frame)
        setupCamera()
    }

    required init?(coder: NSCoder) { super.init(coder: coder) }

    private func setupCamera() {
        guard let device = AVCaptureDevice.default(for: .video),
              let input = try? AVCaptureDeviceInput(device: device) else { return }
        session.addInput(input)

        let output = AVCaptureMetadataOutput()
        session.addOutput(output)
        output.setMetadataObjectsDelegate(self, queue: .main)
        output.metadataObjectTypes = [.qr]

        let layer = AVCaptureVideoPreviewLayer(session: session)
        layer.videoGravity = .resizeAspectFill
        self.layer.addSublayer(layer)
        self.previewLayer = layer

        DispatchQueue.global(qos: .userInitiated).async { self.session.startRunning() }
    }

    override func layoutSubviews() {
        super.layoutSubviews()
        previewLayer?.frame = bounds
    }

    func metadataOutput(_ output: AVCaptureMetadataOutput, didOutput objects: [AVMetadataObject], from connection: AVCaptureConnection) {
        guard !hasScanned,
              let obj = objects.first as? AVMetadataMachineReadableCodeObject,
              let str = obj.stringValue else { return }
        hasScanned = true
        session.stopRunning()
        UINotificationFeedbackGenerator().notificationOccurred(.success)
        onQRCode?(str)
    }
}
```

```swift
import ActivityKit
import SwiftUI

/// Manages the ActivityKit Live Activity lifecycle for connection status.
/// Provides Dynamic Island display on supported devices.
@MainActor
final class KeydLiveActivityManager: ObservableObject {
    private var activity: Activity<KeydConnectionAttributes>?

    func start(status: KeydConnectionAttributes.ContentState.Status) {
        guard ActivityAuthorizationInfo().areActivitiesEnabled else {
            print("[LiveActivity] Live Activities not enabled on this device")
            return
        }

        let initialState = KeydConnectionAttributes.ContentState(status: status, latencyMs: nil)
        let attributes = KeydConnectionAttributes()

        do {
            activity = try Activity<KeydConnectionAttributes>.request(
                attributes: attributes,
                contentState: initialState,
                pushType: nil
            )
            print("[LiveActivity] Started: \(status.rawValue)")
        } catch {
            print("[LiveActivity] Failed to start: \(error)")
        }
    }

    func update(status: KeydConnectionAttributes.ContentState.Status, latencyMs: Int? = nil) {
        let newState = KeydConnectionAttributes.ContentState(status: status, latencyMs: latencyMs)
        Task {
            await activity?.update(using: newState)
        }
    }

    func end() {
        let finalState = KeydConnectionAttributes.ContentState(status: .disconnected, latencyMs: nil)
        Task {
            await activity?.end(using: finalState, dismissalPolicy: .immediate)
            activity = nil
        }
    }
}
```
