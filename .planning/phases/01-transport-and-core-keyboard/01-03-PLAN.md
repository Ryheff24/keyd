import Network
import Foundation
import SharedProtocol

/// Connects to the macOS agent via USB using usbmuxd TCP tunnel.
/// usbmuxd on macOS forwards this iOS device's connections to localhost:12345.
actor USBTransport {
    static let port: UInt16 = 12345
    private var connection: NWConnection?
    var onMessage: ((KeydMessage) -> Void)?
    var onDisconnect: (() -> Void)?

    private let decoder = MessageDecoder()
    private let encoder = MessageEncoder()

    /// Connect with exponential backoff retry.
    func connect() async {
        decoder.onMessage = { [weak self] msg in
            Task { await self?.onMessage?(msg) }
        }

        var delay: TimeInterval = 0.5
        let maxDelay: TimeInterval = 5.0
        var attempts = 0
        let maxAttempts = 10

        while attempts < maxAttempts {
            do {
                try await attemptConnect()
                print("[USBTransport] Connected on attempt \(attempts + 1)")
                return
            } catch {
                attempts += 1
                print("[USBTransport] Connect attempt \(attempts) failed: \(error). Retrying in \(delay)s...")
                try? await Task.sleep(nanoseconds: UInt64(delay * 1_000_000_000))
                delay = min(delay * 2, maxDelay)
            }
        }
        print("[USBTransport] Max retries reached. Connection failed.")
        onDisconnect?()
    }

    private func attemptConnect() async throws {
        let params = NWParameters.tcp
        let tcpOptions = NWProtocolTCP.Options()
        tcpOptions.enableKeepalive = true
        tcpOptions.keepaliveIdle = 10
        params.defaultProtocolStack.transportProtocol = tcpOptions

        let endpoint = NWEndpoint.hostPort(
            host: "127.0.0.1",
            port: NWEndpoint.Port(rawValue: Self.port)!
        )
        let conn = NWConnection(to: endpoint, using: params)
        self.connection = conn

        try await withCheckedThrowingContinuation { (cont: CheckedContinuation<Void, Error>) in
            conn.stateUpdateHandler = { [weak self] state in
                switch state {
                case .ready:
                    cont.resume()
                    Task { await self?.startReceiving(conn) }
                case .failed(let error):
                    cont.resume(throwing: error)
                case .cancelled:
                    cont.resume(throwing: CancellationError())
                default:
                    break
                }
            }
            conn.start(queue: .global(qos: .userInteractive))
        }
    }

    private func startReceiving(_ connection: NWConnection) async {
        while true {
            do {
                let data = try await receiveOnce(on: connection)
                decoder.receive(data)
            } catch {
                print("[USBTransport] Receive error: \(error)")
                onDisconnect?()
                return
            }
        }
    }

    private func receiveOnce(on connection: NWConnection) async throws -> Data {
        try await withCheckedThrowingContinuation { cont in
            connection.receive(minimumIncompleteLength: 1, maximumLength: 65536) { data, _, isComplete, error in
                if let error = error {
                    cont.resume(throwing: error)
                } else if isComplete {
                    cont.resume(throwing: URLError(.networkConnectionLost))
                } else if let data = data {
                    cont.resume(returning: data)
                }
            }
        }
    }

    func send(_ message: KeydMessage) async throws {
        guard let conn = connection else {
            throw URLError(.notConnectedToInternet)
        }
        let frame = try encoder.encode(message)
        try await withCheckedThrowingContinuation { cont in
            conn.send(content: frame, completion: .contentProcessed { error in
                if let error = error {
                    cont.resume(throwing: error)
                } else {
                    cont.resume()
                }
            })
        }
    }

    func disconnect() {
        connection?.cancel()
        connection = nil
    }
}

import Network
import Foundation
import SharedProtocol

/// Discovers and connects to the macOS agent over AWDL via Bonjour (_keyd._tcp).
/// Pre-warmed concurrently with USB; promoted to primary on USB disconnect.
actor AWDLTransport {
    private var browser: NWBrowser?
    private var connection: NWConnection?
    var onMessage: ((KeydMessage) -> Void)?
    var onDisconnect: (() -> Void)?
    var onConnectionReady: (() -> Void)?  // Called when AWDL is ready (pre-warm complete)

    private let decoder = MessageDecoder()
    private let encoder = MessageEncoder()

    func startBrowsing() {
        decoder.onMessage = { [weak self] msg in
            Task { await self?.onMessage?(msg) }
        }

        let params = NWParameters.applicationService
        let browser = NWBrowser(for: .bonjour(type: "_keyd._tcp", domain: nil), using: params)

        browser.browseResultsChangedHandler = { [weak self] results, changes in
            // Connect to first discovered result
            if let result = results.first, self != nil {
                Task { await self?.connectToEndpoint(result.endpoint) }
            }
        }
        browser.stateUpdateHandler = { state in
            switch state {
            case .ready:
                print("[AWDLTransport] Browser ready, searching for _keyd._tcp")
            case .failed(let error):
                print("[AWDLTransport] Browser failed: \(error)")
            default:
                break
            }
        }
        browser.start(queue: .global(qos: .userInteractive))
        self.browser = browser
    }

    private func connectToEndpoint(_ endpoint: NWEndpoint) async {
        guard connection == nil else { return }  // Already connected

        let params = NWParameters.applicationService
        let tcpOptions = NWProtocolTCP.Options()
        tcpOptions.enableKeepalive = true
        params.defaultProtocolStack.transportProtocol = tcpOptions

        let conn = NWConnection(to: endpoint, using: params)
        conn.stateUpdateHandler = { [weak self] state in
            switch state {
            case .ready:
                print("[AWDLTransport] Connected via AWDL")
                Task {
                    await self?.setConnection(conn)
                    await self?.onConnectionReady?()
                    await self?.startReceiving(conn)
                }
            case .failed(let error):
                print("[AWDLTransport] Connection failed: \(error)")
                Task { await self?.setConnection(nil) }
            default:
                break
            }
        }
        conn.start(queue: .global(qos: .userInteractive))
    }

    private func setConnection(_ conn: NWConnection?) async {
        connection = conn
    }

    private func startReceiving(_ conn: NWConnection) async {
        while true {
            do {
                let data = try await receiveOnce(on: conn)
                decoder.receive(data)
            } catch {
                print("[AWDLTransport] Receive error: \(error)")
                connection = nil
                onDisconnect?()
                return
            }
        }
    }

    private func receiveOnce(on connection: NWConnection) async throws -> Data {
        try await withCheckedThrowingContinuation { cont in
            connection.receive(minimumIncompleteLength: 1, maximumLength: 65536) { data, _, isComplete, error in
                if let error = error { cont.resume(throwing: error) }
                else if isComplete { cont.resume(throwing: URLError(.networkConnectionLost)) }
                else if let data = data { cont.resume(returning: data) }
            }
        }
    }

    func send(_ message: KeydMessage) async throws {
        guard let conn = connection else {
            throw URLError(.notConnectedToInternet)
        }
        let frame = try encoder.encode(message)
        try await withCheckedThrowingContinuation { cont in
            conn.send(content: frame, completion: .contentProcessed { error in
                if let error = error { cont.resume(throwing: error) }
                else { cont.resume() }
            })
        }
    }

    var isConnected: Bool { connection != nil }

    func stop() {
        browser?.cancel()
        connection?.cancel()
        browser = nil
        connection = nil
    }
}

import Foundation
import SharedProtocol

/// iOS-side transport state machine.
/// USB primary; AWDL pre-warmed and promoted on USB disconnect (TRANS-03, TRANS-04).
@MainActor
final class TransportManager: ObservableObject {
    enum State {
        case disconnected
        case connecting
        case usbConnected
        case awdlConnected
    }

    @Published var connectionState: State = .disconnected

    private var usbTransport = USBTransport()
    private var awdlTransport = AWDLTransport()

    var onSecureInputChanged: ((Bool) -> Void)?

    // MARK: - Start

    func start() {
        Task {
            await startAWDLPrewarm()
            await startUSBConnection()
        }
    }

    private func startAWDLPrewarm() async {
        await awdlTransport.setOnDisconnect { [weak self] in
            Task { @MainActor in
                if self?.connectionState == .awdlConnected {
                    self?.connectionState = .disconnected
                    self?.start()  // Restart both
                }
            }
        }
        await awdlTransport.setOnConnectionReady { [weak self] in
            Task { @MainActor in
                // AWDL is ready but only promote if USB is not active
                if self?.connectionState == .disconnected || self?.connectionState == .connecting {
                    self?.connectionState = .awdlConnected
                    self?.setIdleTimerState(true)
                }
            }
        }
        await awdlTransport.setOnMessage { [weak self] message in
            Task { @MainActor in
                self?.handleIncomingMessage(message)
            }
        }
        await awdlTransport.startBrowsing()
    }

    private func startUSBConnection() async {
        connectionState = .connecting
        setIdleTimerState(false)

        await usbTransport.setOnDisconnect { [weak self] in
            Task { @MainActor in
                print("[TransportManager] USB disconnected — falling back to AWDL")
                let awdlReady = await self?.awdlTransport.isConnected ?? false
                if awdlReady {
                    self?.connectionState = .awdlConnected
                } else {
                    self?.connectionState = .disconnected
                    self?.setIdleTimerState(false)
                }
            }
        }
        await usbTransport.setOnMessage { [weak self] message in
            Task { @MainActor in
                self?.handleIncomingMessage(message)
            }
        }

        await usbTransport.connect()
        connectionState = .usbConnected
        setIdleTimerState(true)
    }

    // MARK: - Send

    func send(_ message: KeydMessage) {
        Task {
            do {
                switch connectionState {
                case .usbConnected:
                    try await usbTransport.send(message)
                case .awdlConnected:
                    try await awdlTransport.send(message)
                default:
                    print("[TransportManager] Cannot send — not connected")
                }
            } catch {
                print("[TransportManager] Send error: \(error)")
            }
        }
    }

    // MARK: - Receive

    private func handleIncomingMessage(_ message: KeydMessage) {
        switch message {
        case .secureInputChanged(let ev):
            onSecureInputChanged?(ev.isActive)
        case .pairingResponse, .reflectState, .appStateChanged:
            // Handled by PairingManager (Plan 04) or future plans
            break
        default:
            print("[TransportManager] Unexpected iOS-received message: \(message)")
        }
    }

    // MARK: - Idle Timer

    private func setIdleTimerState(_ active: Bool) {
        UIApplication.shared.isIdleTimerDisabled = active
    }
}

// MARK: - AWDLTransport helper for callback setting
extension AWDLTransport {
    func setOnDisconnect(_ handler: @escaping () -> Void) async { onDisconnect = handler }
    func setOnConnectionReady(_ handler: @escaping () -> Void) async { onConnectionReady = handler }
    func setOnMessage(_ handler: @escaping (KeydMessage) -> Void) async { onMessage = handler }
}
extension USBTransport {
    func setOnDisconnect(_ handler: @escaping () -> Void) async { onDisconnect = handler }
    func setOnMessage(_ handler: @escaping (KeydMessage) -> Void) async { onMessage = handler }
}

/// Full-width SwiftUI keyboard mimicking native iOS keyboard appearance.
/// Renders character keys, space bar, and control keys (Backspace, Return).
/// Does NOT use UIInputViewController — this is embedded in a standard View.
struct KeyboardView: View {
    let onKeyTap: (UInt32) -> Void           // Unicode scalar
    let onVirtualKeyTap: (UInt16) -> Void    // Virtual key code (VirtualKey constants)
    let onBackspaceStart: () -> Void
    let onBackspaceEnd: () -> Void

    @State private var pressedKey: String? = nil
    @State private var isShifted = false
    @State private var isCapsLocked = false

    // Standard QWERTY layout rows
    private let rows: [[String]] = [
        ["q","w","e","r","t","y","u","i","o","p"],
        ["a","s","d","f","g","h","j","k","l"],
        ["z","x","c","v","b","n","m"]
    ]

    var body: some View {
        VStack(spacing: 8) {
            // Row 1
            keyRow(rows[0])
            // Row 2 (indented)
            keyRow(rows[1], indent: true)
            // Row 3 (with shift + backspace)
            HStack(spacing: 6) {
                shiftKey
                keyRow(rows[2], spacing: 6)
                backspaceKey
            }
            .padding(.horizontal, 4)
            // Bottom row: 123 | space | return
            HStack(spacing: 6) {
                numberModeKey
                spaceKey
                returnKey
            }
            .padding(.horizontal, 4)
        }
        .padding(.horizontal, 3)
        .padding(.vertical, 8)
        .background(Color(UIColor.systemBackground).opacity(0.001)) // Transparent for Glass layer above
    }

    // MARK: - Key rows

    private func keyRow(_ keys: [String], indent: Bool = false, spacing: CGFloat = 6) -> some View {
        HStack(spacing: spacing) {
            if indent { Spacer().frame(width: 20) }
            ForEach(keys, id: \.self) { key in
                characterKey(key)
            }
            if indent { Spacer().frame(width: 20) }
        }
        .padding(.horizontal, 3)
    }

    private func characterKey(_ key: String) -> some View {
        let displayChar = (isShifted || isCapsLocked) ? key.uppercased() : key
        return Button(action: {
            let scalar = (displayChar.unicodeScalars.first?.value) ?? 0
            onKeyTap(scalar)
            // Auto-unshift after one key (unless caps lock)
            if isShifted && !isCapsLocked { isShifted = false }
            UIImpactFeedbackGenerator(style: .light).impactOccurred()
        }) {
            Text(displayChar)
                .font(.system(size: 17, weight: .regular))
                .frame(maxWidth: .infinity)
                .frame(height: 42)
                .background(
                    RoundedRectangle(cornerRadius: 5)
                        .fill(Color(UIColor.systemBackground))
                        .shadow(color: .black.opacity(0.3), radius: 0, x: 0, y: 1)
                )
                .overlay(
                    RoundedRectangle(cornerRadius: 5)
                        .stroke(Color.gray.opacity(0.2), lineWidth: 0.5)
                )
        }
        .buttonStyle(.plain)
    }

    // MARK: - Special keys

    private var shiftKey: some View {
        Button(action: {
            if isCapsLocked { isCapsLocked = false; isShifted = false }
            else if isShifted { isCapsLocked = true }
            else { isShifted = true }
            UIImpactFeedbackGenerator(style: .light).impactOccurred()
        }) {
            Image(systemName: isCapsLocked ? "capslock.fill" : (isShifted ? "shift.fill" : "shift"))
                .frame(width: 42, height: 42)
                .background(RoundedRectangle(cornerRadius: 5).fill(Color(UIColor.systemBackground)))
        }
        .buttonStyle(.plain)
    }

    private var backspaceKey: some View {
        // CONTEXT.md locked decision: tap → single char delete; hold 0.5s → word delete (Cmd-Backspace)
        // Phase 1 implements both: single char on tap, word delete (HotkeyEvent with .command modifier) on long press
        Button(action: {
            onVirtualKeyTap(VirtualKey.delete)
            UIImpactFeedbackGenerator(style: .light).impactOccurred()
        }) {
            Image(systemName: "delete.left")
                .frame(width: 42, height: 42)
                .background(RoundedRectangle(cornerRadius: 5).fill(Color(UIColor.systemBackground)))
        }
        .buttonStyle(.plain)
        .onLongPressGesture(minimumDuration: 0.5, perform: {
            // After holding 0.5s: switch to word delete via Cmd-Backspace (Option-Delete on macOS)
            // HotkeyEvent with .command modifier sends Cmd-Backspace = delete word to left of cursor
            UIImpactFeedbackGenerator(style: .medium).impactOccurred()
            onBackspaceStart()  // Signal continuous word-delete mode begins
        })
        .simultaneousGesture(
            DragGesture(minimumDistance: 0)
                .onEnded { _ in onBackspaceEnd() }
        )
    }

    private var spaceKey: some View {
        Button(action: {
            onKeyTap(32)  // Space
            UIImpactFeedbackGenerator(style: .light).impactOccurred()
        }) {
            Text("space")
                .font(.system(size: 15))
                .foregroundColor(.secondary)
                .frame(maxWidth: .infinity)
                .frame(height: 42)
                .background(RoundedRectangle(cornerRadius: 5).fill(Color(UIColor.systemBackground)))
        }
        .buttonStyle(.plain)
    }

    private var returnKey: some View {

import SwiftUI
import SharedProtocol

/// Root view: connection status badge at top, hotkey panel, then keyboard at bottom.
/// Full-screen keyboard layout, portrait only.
struct RootView: View {
    @EnvironmentObject var transportManager: TransportManager
    @State private var isSecureInputActive = false
    @State private var showQuitConfirmation = false
    @State private var pendingDestructiveHotkey: (UInt16, ModifierFlags)? = nil

    var body: some View {
        ZStack {
            // Dark background to make Liquid Glass panels pop
            Color(UIColor.systemBackground).ignoresSafeArea()

            VStack(spacing: 0) {
                // Status indicator at top (fallback for non-Dynamic Island devices)
                HStack {
                    Spacer()
                    ConnectionStatusBadge(state: transportManager.connectionState)
                    Spacer()
                }
                .padding(.top, 8)

                Spacer()

                // Secure input overlay
                if isSecureInputActive {
                    secureInputOverlay
                } else {
                    keyboardStack
                }
            }
            .padding(.bottom, 8)
        }
        .ignoresSafeArea(.keyboard)
        .onReceive(transportManager.$connectionState) { state in
            // Update idle timer when connection state changes
        }
        .alert("Send Quit Command?", isPresented: $showQuitConfirmation) {
            Button("Send ⌘Q", role: .destructive) {
                if let (key, mods) = pendingDestructiveHotkey {
                    sendHotkey(virtualKey: key, modifiers: mods)
                }
                pendingDestructiveHotkey = nil
            }
            Button("Cancel", role: .cancel) {
                pendingDestructiveHotkey = nil
            }
        } message: {
            Text("This will quit the frontmost macOS app.")
        }
        .onReceive(
            NotificationCenter.default.publisher(for: UIApplication.didBecomeActiveNotification)
        ) { _ in
            transportManager.onSecureInputChanged = { active in
                isSecureInputActive = active
            }
        }
    }

    private var keyboardStack: some View {
        VStack(spacing: 4) {
            // Hotkey panel above keyboard (UX-03)
            HotkeyPanel(
                onHotkey: { key, mods in sendHotkey(virtualKey: key, modifiers: mods) },
                onDestructiveHotkey: { key, mods, label in
                    pendingDestructiveHotkey = (key, mods)
                    showQuitConfirmation = true
                }
            )

            // Main keyboard (UX-03)
            KeyboardView(
                onKeyTap: { scalar in
                    transportManager.send(.keyEvent(KeyEvent(unicodeScalar: scalar, keyDown: true)))
                    transportManager.send(.keyEvent(KeyEvent(unicodeScalar: scalar, keyDown: false)))
                },
                onVirtualKeyTap: { keyCode in
                    transportManager.send(.hotkeyEvent(HotkeyEvent(virtualKeyCode: keyCode, keyDown: true)))
                    transportManager.send(.hotkeyEvent(HotkeyEvent(virtualKeyCode: keyCode, keyDown: false)))
                },
                onBackspaceStart: {
                    // Long-press (0.5s) entered word-delete mode: send Cmd-Backspace
                    transportManager.send(.hotkeyEvent(HotkeyEvent(virtualKeyCode: VirtualKey.delete, keyDown: true, modifiers: .command)))
                    transportManager.send(.hotkeyEvent(HotkeyEvent(virtualKeyCode: VirtualKey.delete, keyDown: false, modifiers: .command)))
                },
                onBackspaceEnd: {
                    // No-op: word-delete is sent as a single atomic event in onBackspaceStart
                    // (unlike char-delete which is continuous — word-delete fires once per hold trigger)
                }
            )
        }
    }

    private var secureInputOverlay: some View {
        VStack(spacing: 12) {
            Image(systemName: "lock.fill")
                .font(.system(size: 40))
                .foregroundColor(.secondary)
            Text("Password Field Active")
                .font(.headline)
            Text("Keyboard input paused while a password field is focused on your Mac.")
                .font(.caption)
                .multilineTextAlignment(.center)
                .foregroundColor(.secondary)
        }
        .padding()
        .frame(maxWidth: .infinity)
        .background(.ultraThinMaterial)
        .cornerRadius(16)
        .padding()
    }

    private func sendHotkey(virtualKey: UInt16, modifiers: ModifierFlags) {
        // Send key-down with modifiers; the macOS side uses defer for key-up
        transportManager.send(.hotkeyEvent(HotkeyEvent(virtualKeyCode: virtualKey, keyDown: true, modifiers: modifiers)))
        transportManager.send(.hotkeyEvent(HotkeyEvent(virtualKeyCode: virtualKey, keyDown: false, modifiers: modifiers)))
    }
}
